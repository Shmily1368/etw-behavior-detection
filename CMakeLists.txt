cmake_minimum_required(VERSION 3.20)
project(etw_behavior_detection VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(behavior_features collector/src/feature_aggregator.cpp)
target_include_directories(behavior_features PUBLIC collector/include)
if(MSVC)
  target_compile_options(behavior_features PRIVATE /W4 /utf-8)
else()
  target_compile_options(behavior_features PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(feature_replay collector/src/feature_replay.cpp)
target_link_libraries(feature_replay PRIVATE behavior_features)
if(BUILD_TESTING)
  add_executable(feature_contract_test collector/tests/feature_contract_test.cpp)
  target_link_libraries(feature_contract_test PRIVATE behavior_features)
  add_test(NAME feature_contract COMMAND feature_contract_test)
endif()

if(WIN32)
  add_executable(etw_capture collector/src/etw_capture.cpp)
  target_compile_definitions(etw_capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(etw_capture PRIVATE /W4 /utf-8)
  target_link_libraries(etw_capture PRIVATE advapi32)
  add_test(NAME collector_help COMMAND etw_capture --help)
endif()
